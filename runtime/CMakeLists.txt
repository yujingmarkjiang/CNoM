cmake_minimum_required(VERSION 3.16)
project(cnom_runtime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(cnom STATIC src/gate.cpp src/router.cpp)
target_include_directories(cnom PUBLIC include)
add_executable(cnom-edge src/edge_main.cpp)
target_link_libraries(cnom-edge PRIVATE cnom)
add_executable(cnom-runtime-test tests/test_runtime.cpp)
target_link_libraries(cnom-runtime-test PRIVATE cnom)
enable_testing()
add_test(NAME runtime COMMAND cnom-runtime-test)
